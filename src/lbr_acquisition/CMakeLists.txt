cmake_minimum_required(VERSION 3.22)
project(lbr_acquisition_offline VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf 3.21.12 EXACT REQUIRED)
execute_process(COMMAND "${Protobuf_PROTOC_EXECUTABLE}" --version
  OUTPUT_VARIABLE PROTOC_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE
  RESULT_VARIABLE PROTOC_RESULT)
if(NOT PROTOC_RESULT EQUAL 0 OR NOT PROTOC_VERSION STREQUAL "libprotoc 3.21.12")
  message(FATAL_ERROR "Offline experiment requires protoc 3.21.12")
endif()
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(MCAP_SOURCE_DIR "" CACHE PATH "Pinned MCAP source checkout (not the ROS vendor)")
set(MCAP_INCLUDE "${MCAP_SOURCE_DIR}/cpp/mcap/include")
if(NOT EXISTS "${MCAP_INCLUDE}/mcap/types.hpp")
  message(FATAL_ERROR "Set MCAP_SOURCE_DIR to pinned MCAP source")
endif()
file(READ "${MCAP_INCLUDE}/mcap/types.hpp" MCAP_TYPES)
if(NOT MCAP_TYPES MATCHES "#define MCAP_LIBRARY_VERSION \"2.1.3\"")
  message(FATAL_ERROR "Offline experiment requires MCAP C++ 2.1.3")
endif()
# Fail if any pinned native header was edited or came from a different release.
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/mcap-2.1.3.sha256" MCAP_HASHES)
foreach(entry IN LISTS MCAP_HASHES)
  string(SUBSTRING "${entry}" 0 64 expected)
  string(SUBSTRING "${entry}" 66 -1 relative)
  file(SHA256 "${MCAP_INCLUDE}/${relative}" actual)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "Pinned MCAP source mismatch: ${relative}")
  endif()
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/mcap-source.txt" "commit=00e47a1ba6901cbc0cf448cccebbc554c545c4fa\nall native header hashes verified\nsource=${MCAP_SOURCE_DIR}\n")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
add_custom_command(OUTPUT "${GEN}/acquisition_v1.pb.cc" "${GEN}/acquisition_v1.pb.h" "${GEN}/acquisition_v1_pb2.py" "${GEN}/acquisition_v1.desc"
 COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${GEN} --python_out=${GEN} --include_imports --descriptor_set_out=${GEN}/acquisition_v1.desc ${CMAKE_CURRENT_SOURCE_DIR}/proto/acquisition_v1.proto
 DEPENDS proto/acquisition_v1.proto VERBATIM)
add_library(acquisition_offline src/storage/capture_encoding.cpp src/storage/offline_writer.cpp src/storage/readback.cpp "${GEN}/acquisition_v1.pb.cc")
target_include_directories(acquisition_offline PUBLIC include "${GEN}")
target_include_directories(acquisition_offline SYSTEM PRIVATE "${MCAP_INCLUDE}")
target_compile_definitions(acquisition_offline PRIVATE MCAP_COMPRESSION_NO_LZ4 MCAP_COMPRESSION_NO_ZSTD LBR_BUILD_DESCRIPTOR="${GEN}/acquisition_v1.desc")
target_link_libraries(acquisition_offline PUBLIC protobuf::libprotobuf PRIVATE OpenSSL::Crypto)
target_compile_options(acquisition_offline PRIVATE -Wall -Wextra -Wpedantic)
add_executable(offline_fixture src/offline_fixture.cpp)
target_link_libraries(offline_fixture PRIVATE acquisition_offline)
find_package(Boost 1.83 EXACT REQUIRED)
find_package(Threads REQUIRED)
file(SHA256 "${Boost_INCLUDE_DIRS}/boost/lockfree/spsc_queue.hpp" BOOST_SPSC_HASH)
if(NOT BOOST_SPSC_HASH STREQUAL "0540de5f6b2403b9ef07c4ecfd6297e89cd994059295c599876d82a66ddbc485")
  message(FATAL_ERROR "Offline core requires the reviewed Boost 1.83 SPSC header")
endif()
add_library(acquisition_capture src/recorder.cpp)
target_include_directories(acquisition_capture PUBLIC include)
target_link_libraries(acquisition_capture PUBLIC Boost::headers)
target_compile_options(acquisition_capture PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
